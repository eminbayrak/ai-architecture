**Goal:** Public-safe local kit at `~/projects/fde-lab` with a runnable FDE LangGraph crew plus portable `transcribe-video` + `watch` skills. Do not `gh repo create` or `git push` until Emin says yes.

- [ ] **Step 3: Write `agents/graphs/fde_crew.py`**

```python
from __future__ import annotations

import os
import sys
from pathlib import Path
from typing import TypedDict

from langchain_core.messages import HumanMessage, SystemMessage
from langgraph.graph import END, START, StateGraph

PROMPTS_DIR = Path(__file__).resolve().parents[1] / "prompts"
NODE_FIELDS = (
    ("intake", "engagement_brief"),
    ("research", "research_brief"),
    ("eval", "eval_plan"),
    ("delivery", "delivery"),
)


class CrewState(TypedDict):
    customer_ask: str
    engagement_brief: str
    research_brief: str
    eval_plan: str
    delivery: str
    error: str


def load_prompt(name: str) -> str:
    path = PROMPTS_DIR / f"{name}.md"
    return path.read_text(encoding="utf-8")


def require_api_key() -> None:
    if not os.environ.get("OPENAI_API_KEY"):
        print("Missing OPENAI_API_KEY. Copy .env.example to .env and set the key.", file=sys.stderr)
        raise SystemExit(1)


def _empty_state(customer_ask: str) -> CrewState:
    return {
        "customer_ask": customer_ask,
        "engagement_brief": "",
        "research_brief": "",
        "eval_plan": "",
        "delivery": "",
        "error": "",
    }


def _context(state: CrewState) -> str:
    return (
        f"Customer ask:\n{state['customer_ask']}\n\n"
        f"Engagement brief:\n{state['engagement_brief']}\n\n"
        f"Research brief:\n{state['research_brief']}\n\n"
        f"Eval plan:\n{state['eval_plan']}\n"
    )


def _make_node(name: str, field: str, model):
    prompt = load_prompt(name)

    def node(state: CrewState) -> dict:
        if state.get("error"):
            return {}
        try:
            result = model.invoke(
                [SystemMessage(content=prompt), HumanMessage(content=_context(state))]
            )
            text = result.content if hasattr(result, "content") else str(result)
            return {field: text}
        except Exception as exc:
            return {"error": f"{name} failed: {exc}"}

    node.__name__ = name
    return node


def _route(state: CrewState) -> str:
    return "end" if state.get("error") else "next"


def build_graph(model):
    graph = StateGraph(CrewState)
    for name, field in NODE_FIELDS:
        graph.add_node(name, _make_node(name, field, model))
    graph.add_edge(START, "intake")
    graph.add_conditional_edges("intake", _route, {"next": "research", "end": END})
    graph.add_conditional_edges("research", _route, {"next": "eval", "end": END})
    graph.add_conditional_edges("eval", _route, {"next": "delivery", "end": END})
    graph.add_edge("delivery", END)
    return graph.compile()


def run_crew(customer_ask: str, model) -> CrewState:
    graph = build_graph(model)
    return graph.invoke(_empty_state(customer_ask))


def _live_model():
    from langchain_openai import ChatOpenAI

    kwargs = {"model": os.environ.get("FDE_CREW_MODEL", "gpt-4.1-mini")}
    base = os.environ.get("OPENAI_BASE_URL")
    if base:
        kwargs["base_url"] = base
    return ChatOpenAI(**kwargs)


def main() -> None:
    require_api_key()
    ask = " ".join(sys.argv[1:]).strip()
    if not ask:
        print("Usage: uv run fde-crew \"customer wants X under constraint Y\"", file=sys.stderr)
        raise SystemExit(1)
    state = run_crew(ask, _live_model())
    sections = (
        ("intake", "engagement_brief"),
        ("research", "research_brief"),
        ("eval", "eval_plan"),
        ("delivery", "delivery"),
    )
    for label, field in sections:
        print(f"## {label}\n")
        print(state.get(field) or "")
        print()
        if state.get("error"):
            print(f"error: {state['error']}", file=sys.stderr)
            raise SystemExit(1)
    if state.get("error"):
        print(f"error: {state['error']}", file=sys.stderr)
        raise SystemExit(1)
```

Fix `main()` so it prints completed sections, then exits 1 if `error` is set (print error last). Do not check error inside the print loop in a way that duplicates. Final `main` error handling:

        print(state.get(field) or "")
        print()
